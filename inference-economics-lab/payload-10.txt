Output'), nonInf=num('#econNonInference')/100;
    const lifeSeconds=life*365.25*24*3600;
    const depreciationActive=hw/(lifeSeconds*util);
    const energySecond=power*pue*elec/3600;
    const directSecond=(depreciationActive+energySecond)*ops;
    const inputCogs=directSecond*(1e6/prefill);
    const outputCogs=directSecond*(1e6/decode);
    const blendedRevenue=soldIn+soldOut;
    const blendedCogs=inputCogs+outputCogs;
    const margin=blendedRevenue>0?1-blendedCogs/blendedRevenue:0;
    const fullyLoaded=blendedCogs+blendedRevenue*nonInf;
    const retained=blendedRevenue-fullyLoaded;
    $('#econCostSecond').textContent=fmtMoney(directSecond,5);
    $('#econInputCogs').textContent=fmtMoney(inputCogs,2);
    $('#econOutputCogs').textContent=fmtMoney(outputCogs,2);
    $('#econMargin').textContent=fmtPct(margin,0);
    const rows=[['Direct serving COGS',blendedCogs],['Non-inference allocation',blendedRevenue*nonInf],['Retained contribution',Math.max(0,retained)],['Commercial price',blendedRevenue]];
    const max=Math.max(...rows.map(r=>r[1]),.001);
    $('#econStack').innerHTML=rows.map(r=>`<div class="waterfall-row"><span>${r[0]}</span><div class="waterfall-track"><div class="waterfall-fill" style="width:${clamp(r[1]/max*100,0,100)}%"></div></div><strong>${fmtMoney(r[1])}</strong></div>`).join('');
    let text=`At these assumptions, decode costs about ${inputCogs>0?(outputCogs/inputCogs).toFixed(1):'—'}× as much per token as prefill because aggregate decode throughput is lower. `;
    if (retained >= 0) text += `${fmtMoney(retained)} per paired million input + output tokens remains after the chosen direct and non-inference allocations.`;
    else text += `The assumed tariff does not cover the chosen serving and non-inference allocations; the shortfall is ${fmtMoney(-retained)} per paired million input + output tokens.`;
    text += ' This does not imply a real provider margin: actual throughput, utilization, negotiated prices, and cost allocation are private.';
    $('#econInterpretation').textContent=text;
  }

  function updateSubscription() {
    const plan=num('#subPrice'), days=num('#subDays'), sessions=num('#subSessions'), turns=num('#subTurns'), input=num('#subInput'), output=num('#subOutput'), ip=num('#subInputPrice'), op=num('#subOutputPrice');
    const monthlyTurns=days*sessions*turns;
    const apiPerTurn=input/1e6*ip+output/1e6*op;
    const api=monthlyTurns*apiPerTurn;
    const diff=api-plan;
    const be=apiPerTurn>0?plan/apiPerTurn:Infinity;
    $('#subApiEquivalent').textContent=fmtMoney(api);
    $('#subDifference').textContent=diff>=0?`${fmtMoney(diff)} less on plan`:`${fmtMoney(-diff)} premium`;
    $('#subBreakEven').textContent=Number.isFinite(be)?fmtNum(be):'—';
    $('#subPerTurn').textContent=monthlyTurns>0?fmtMoney(plan/monthlyTurns,3):'—';
    let insight;
    if(api>plan*1.5) insight='At this workload, the subscription appears valuable relative to list-price API use—but the comparison assumes the product lets you sustain this usage, use equivalent models, and avoid queueing or weekly limits.';
    else if(api<plan*.67) insight='The API appears cheaper at the stated workload. A subscription may still be valuable for bundled tools, product UX, privacy controls, or unpredictable spikes.';
    else insight='The direct price comparison is close. Workflow fit, model access, caching behaviour, limits, and bundled tools are likely more important than the headline fee.';
    $('#subInsight').textContent=insight;
  }
